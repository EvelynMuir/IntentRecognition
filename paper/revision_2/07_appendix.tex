\section{Notation}
\label{app:notation}
Table~\ref{tab:notation} summarizes the notation used throughout the paper.

\begin{table}[H]
\centering
\small
\caption{Summary of notation.}
\label{tab:notation}
\setlength{\tabcolsep}{6pt}

\begin{tabular}{ll}
\hline
Symbol & Meaning \\
\hline
$x$ & input image \\
$\mathcal{Y},\,C$ & intent label space and number of intent classes \\
$Y\in\{0,1\}^C$ & latent multi-label intent \\
$\mathbf{y}^{\text{soft}}$ & soft annotation vector in $\{0,0.33,0.66,1\}^C$ \\
$\mathbf{y}^{\text{hard}}$ & binarized hard labels \\
$\omega\in(0,1]$ & sample-level annotator agreement score \\
$f_{\text{img}},\,\mathbf{z}=f_{\text{img}}(x)$ & frozen visual (CLIP) encoder and image feature \\
$\mathbf{e}_c,\,\mathbf{e}_{\text{text}}$ & class-text / rationale text embedding \\
$\gamma$ & CLIP logit scale \\
$s^{\text{prior}}_c,\,\tilde{s}^{\text{prior}}_c$ & raw / normalized heterogeneous semantic prior for class $c$ \\
$\mathrm{mean}^{r}(x),\,\mathrm{std}^{r}(x)$ & per-sample statistics of source $r$ over the $C$ classes \\
$\mathbf{s}^{b},\,\mathbf{s}^{v}$ & base / visual-student logits \\
$\mathcal{C}_K(x),\,K$ & Top-$K$ candidate set and its size \\
$\alpha$ & semantic-reranking strength \\
$h_{\text{res}}(\cdot),\,h_{\text{tea}}(\cdot)$ & residual student head, text-teacher head \\
$s^{\text{full}}_c$ & final refined score for class $c$ \\
$t_c$ & class-wise decision threshold (validation-only) \\
$\mathbf{p}^{\text{tea}},\,\tau$ & text-teacher distribution and temperature \\
$g=1-\omega$ & teacher-branch gate weight \\
$\lambda$ & distillation loss weight \\
$\mathcal{L}_{\text{vis}},\mathcal{L}_{\text{distill}},\mathcal{L}_{\text{total}}$ & visual, distillation, and total losses \\
$\eta_v,\,\eta_t$ & annotation / teacher noise in the gate analysis \\
$\sigma_v^2(\omega),\,\sigma_t^2,\,b_t$ & their variances and the teacher bias \\
$X,\,Y_c$ & random variables of the image and the annotation of intent $c$ \\
$Z,\,B,\,S$ & random variables of $\mathbf{z}$, $\mathbf{s}^{b}$, and $\tilde{\mathbf{s}}^{\text{prior}}$ \\
\hline
\end{tabular}

\end{table}
